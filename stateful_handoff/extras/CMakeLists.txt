cmake_minimum_required(VERSION 3.20)
project(stateful_mixed LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(DEP_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../deps/root" CACHE PATH "Protobuf headers/library")
find_path(PB_INCLUDE google/protobuf/message.h HINTS "${DEP_ROOT}/usr/include" REQUIRED)
find_library(PB_LIBRARY protobuf HINTS "${DEP_ROOT}/usr/lib/x86_64-linux-gnu" REQUIRED)
find_package(Threads REQUIRED)
add_executable(mixed_bench mixed_bench.cpp ../events.pb.cc)
target_include_directories(mixed_bench PRIVATE ${PB_INCLUDE} ..)
target_compile_options(mixed_bench PRIVATE -O3 -DNDEBUG -march=native)
target_link_libraries(mixed_bench PRIVATE ${PB_LIBRARY} Threads::Threads)
foreach(program proto_mixed proto_mixed_alloc)
 add_executable(${program} proto_mixed.cpp ../events.pb.cc)
 target_include_directories(${program} PRIVATE ${PB_INCLUDE} ..)
 target_compile_options(${program} PRIVATE -O3 -DNDEBUG -march=native)
 target_link_libraries(${program} PRIVATE ${PB_LIBRARY} Threads::Threads)
endforeach()
target_compile_definitions(proto_mixed_alloc PRIVATE ALLOC_AUDIT)
