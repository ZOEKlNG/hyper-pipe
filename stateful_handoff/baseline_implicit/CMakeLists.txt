cmake_minimum_required(VERSION 3.20)
project(stateful_benchmark LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(DEP_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../deps/root" CACHE PATH "Extracted protobuf development packages")
option(SANITIZE "Build ASan/UBSan functional validation" OFF)
find_package(Threads REQUIRED)
find_path(PB_INCLUDE google/protobuf/message.h HINTS "${DEP_ROOT}/usr/include" REQUIRED)
find_library(PB_LIBRARY protobuf HINTS "${DEP_ROOT}/usr/lib/x86_64-linux-gnu" REQUIRED)
add_library(events STATIC events.pb.cc)
target_include_directories(events PUBLIC ${PB_INCLUDE} ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(events PUBLIC ${PB_LIBRARY} Threads::Threads)
foreach(program bench proto_bench store_audit semantics_demo allocator_scaling)
 if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${program}.cpp")
  add_executable(${program} ${program}.cpp)
  target_link_libraries(${program} PRIVATE events)
 endif()
endforeach()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/proto_bench.cpp")
 add_executable(proto_alloc proto_bench.cpp)
 target_compile_definitions(proto_alloc PRIVATE ALLOC_AUDIT)
 target_link_libraries(proto_alloc PRIVATE events)
endif()
if(SANITIZE)
 add_compile_options(-O1 -g -fno-omit-frame-pointer -fsanitize=address,undefined)
 add_link_options(-fsanitize=address,undefined)
 # Directory options above are also applied explicitly to preexisting targets.
 foreach(t events bench proto_bench proto_alloc store_audit semantics_demo allocator_scaling)
  if(TARGET ${t})
   target_compile_options(${t} PRIVATE -O1 -g -fno-omit-frame-pointer -fsanitize=address,undefined)
   target_link_options(${t} PRIVATE -fsanitize=address,undefined)
  endif()
 endforeach()
else()
 add_compile_options(-O3 -DNDEBUG -march=native)
 foreach(t events bench proto_bench proto_alloc store_audit semantics_demo allocator_scaling)
  if(TARGET ${t})
   target_compile_options(${t} PRIVATE -O3 -DNDEBUG -march=native)
  endif()
 endforeach()
endif()
